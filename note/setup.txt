

## installation
===============
php artisan install:api --passport



## do use HasApiTokens to users model
=====================================
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Foundation\Auth\User as Authenticatable;
use Illuminate\Notifications\Notifiable;
use Laravel\Passport\Contracts\OAuthenticatable;
use Laravel\Passport\HasApiTokens;

class User extends Authenticatable implements OAuthenticatable
{
    use HasApiTokens, HasFactory, Notifiable;
}


## do configuration passport to config/auth.php
===============================================
'guards' => [
    'web' => [
        'driver' => 'session',
        'provider' => 'users',
    ],

    'api' => [
        'driver' => 'passport',
        'provider' => 'users',
    ],
],


## for deploying passport run this for key generate
===================================================
php artisan passport:keys

// this keys will generate on storage folder


## finally run this for create passport client
===============================================
php artisan passport:client --personal
